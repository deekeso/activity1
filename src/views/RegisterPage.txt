// Get the router object

const birthDate = ref('')
const studentID = ref('')

const studentStore = useStudentStore()
// const studentList = computed(() => studentStore.studentList)




//function to get the real age automatically
const getAge = () => {
  const birthDateValue = new Date(birthDate.value)
  const today = new Date()
  let age = today.getFullYear() - birthDateValue.getFullYear()
  const monthDifference = today.getMonth() - birthDateValue.getMonth()

  if ((monthDifference === 0 && today.getDate() < birthDateValue.getDate())) {
    age--;
  }
  return age
}


const formatDate = (dateString: string) => {
  const date = new Date(dateString)
  const month = String(date.getMonth() + 1).padStart(2, '0')
  const day = String(date.getDate()).padStart(2, '0')
  const year = date.getFullYear()
  return `${month}/${day}/${year}`
}

const onSubmit = () => {
  if (firstName.value === '' || middleInitial.value === '' || lastName.value === '' || address.value === '' || birthDate.value === '') {
    console.log('Please fill up all fields')
  } else {
    const newStudent = {
      firstName: firstName.value,
      middleName: middleInitial.value,
      lastName: lastName.value,
      address: address.value,
      birthDate: birthDate.value,
      age: getAge(),
      studentId: 0,
    }
    studentStore.registerStudent(newStudent)
    firstName.value = ''
    middleInitial.value = ''
    lastName.value = ''
    address.value = ''
    birthDate.value = ''
    console.log('Successfully registered')
    console.log('Current student list:', studentStore.studentList)
  }
}


const goToLogin = () => {
  router.push({ name: 'home' })
}
